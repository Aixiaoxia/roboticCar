cmake_minimum_required(VERSION 3.16)
project(bunker_tag_follower)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/follower scripts/velocity_guard scripts/depth_clearance scripts/tag_overlay scripts/control_panel DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(control_policy test/test_policy.py)
  ament_add_pytest_test(depth_policy test/test_depth.py)
  ament_add_pytest_test(planar_kalman test/test_kalman.py)
  ament_add_pytest_test(configuration test/test_configuration.py)
endif()
ament_package()
